 36class SlidingWindowBuffer:
 37    """Short-term memory buffer for recent conversation turns.
 38
 39    Maintains a fixed-size sliding window of the most recent interactions
 40    used to provide immediate conversational context.
 41    """
 42
 43    def __init__(self, max_size: int = 10) -> None:
 44        settings = get_settings()
 45        self._max_size = max_size or settings.agent.sliding_window_size
 46        self._buffer: List[MemoryTurn] = []
 47
 48    def add(self, turn: MemoryTurn) -> None:
 49        """Add a conversation turn to the sliding window buffer."""
 50        self._buffer.append(turn)
 51        if len(self._buffer) > self._max_size:
 52            self._buffer.pop(0)
 53
 54    def get_recent(self, n: Optional[int] = None) -> List[MemoryTurn]:
 55        """Return the most recent conversation turns from the buffer."""
 56        if n is None:
 57            return list(self._buffer)
 58        return list(self._buffer[-n:])
 59
 60    def clear(self) -> None:
 61        """Remove all stored turns from the sliding window buffer."""
 62        self._buffer.clear()
 63
 64    @property
 65    def size(self) -> int:
 66        """Return the current number of conversation turns stored in the sliding window buffer."""
 67        return len(self._buffer)

 70class SemanticMemory:
 71    """Long-term semantic memory backed by a vector database.
 72
 73    Stores conversation turns as embeddings and allows retrieval of
 74    relevant past context using semantic similarity search.
 75    """
 76
 77    def __init__(
 78        self,
 79        collection_name: str = "",
 80        persist_dir: str = "",
 81    ) -> None:
 82        settings = get_settings()
 83        self._collection_name = collection_name or settings.chroma.collection_name
 84        self._persist_dir = persist_dir or settings.chroma.persist_dir
 85
 86        try:
 87            if settings.chroma.use_http:
 88                self._client = chromadb.HttpClient(
 89                    host=settings.chroma.host,
 90                    port=settings.chroma.port,
 91                )
 92            else:
 93                self._client = chromadb.PersistentClient(path=self._persist_dir)
 94
 95            self._collection = self._client.get_or_create_collection(
 96                name=self._collection_name,
 97                metadata={"hnsw:space": "cosine"},
 98            )
 99            logger.info("ChromaDB initialized: collection=%s", self._collection_name)
100        except Exception as exc:
101            logger.warning("ChromaDB initialization failed: %s. Using fallback.", exc)
102            self._client = None
103            self._collection = None
104
105    def store(self, turn: MemoryTurn) -> None:
106        """Persist a conversation turn in semantic memory.
107
108        Stores the turn content and metadata in the configured vector database
109        for later semantic retrieval. If the vector store is unavailable,
110        the operation is silently skipped.
111        """
112        if self._collection is None:
113            return
114        try:
115            self._collection.add(
116                ids=[turn.turn_id],
117                documents=[turn.content],
118                metadatas=[{"role": turn.role, **turn.metadata}],
119            )
120        except Exception as exc:
121            logger.warning("Failed to store in ChromaDB: %s", exc)
122
123    def retrieve(self, query: str, top_k: int = 5) -> List[dict]:
124        """Retrieve semantically similar memory entries.
125
126        Args:
127            query: Natural language query used for similarity search.
128            top_k: Maximum number of results to return.
129
130        Returns:
131            A list of dictionaries containing content, metadata, and similarity distance.
132        """
133        if self._collection is None:
134            return []
135        try:
136            if self._collection.count() == 0:
137                return []
138            n = min(top_k, self._collection.count())
139            results = self._collection.query(query_texts=[query], n_results=n)
140            hits: List[dict] = []
141            if results and results["documents"]:
142                for doc, meta, dist in zip(
143                    results["documents"][0],
144                    results["metadatas"][0],
145                    results["distances"][0],
146                ):
147                    hits.append({"content": doc, "metadata": meta, "distance": dist})
148            return hits
149        except Exception as exc:
150            logger.warning("ChromaDB query failed: %s", exc)
151            return []
152
153    def clear(self) -> None:
154        """Remove all stored semantic memory entries and recreate the collection."""
155        if self._client is None or self._collection is None:
156            return
157        try:
158            self._client.delete_collection(self._collection_name)
159            self._collection = self._client.get_or_create_collection(
160                name=self._collection_name,
161                metadata={"hnsw:space": "cosine"},
162            )
163        except Exception as exc:
164            logger.warning("ChromaDB clear failed: %s", exc)
165
166    @property
167    def count(self) -> int:
168        """Return the total number of stored semantic memory records."""
169        if self._collection is None:
170            return 0
171        try:
172            return self._collection.count()
173        except Exception:
174            return 0

177class AgentMemory:
178    """Combined memory system used by agents.
179
180    Provides both short-term sliding window memory and long-term
181    semantic retrieval for building rich LLM context.
182    """
183
184    def __init__(self) -> None:
185        self.buffer = SlidingWindowBuffer()
186        self.semantic = SemanticMemory()
187
188    def add_turn(self, role: str, content: str, **metadata: str) -> None:
189        """Add a new conversation turn to both short-term and semantic memory.
190
191        The turn is appended to the sliding window buffer and stored in the
192        vector database for semantic retrieval.
193        """
194        turn = MemoryTurn(role=role, content=content, metadata=metadata)
195        self.buffer.add(turn)
196        self.semantic.store(turn)
197
198    def build_context(self, current_query: str, top_k: int = 5) -> str:
199        """Construct a context string combining semantic and recent memory.
200
201        Retrieves relevant past messages from semantic memory and combines them
202        with the recent sliding-window conversation to produce a prompt context
203        for the LLM.
204        """
205        semantic_hits = self.semantic.retrieve(current_query, top_k=top_k)
206        seen_contents: set[str] = set()
207        parts: List[str] = []
208
209        if semantic_hits:
210            parts.append("=== Relevant Past Context (Semantic Memory) ===")
211            for hit in semantic_hits:
212                if hit["content"] not in seen_contents:
213                    role = hit["metadata"].get("role", "unknown")
214                    parts.append(f"[{role}] {hit['content']}")
215                    seen_contents.add(hit["content"])
216
217        recent = self.buffer.get_recent()
218        if recent:
219            parts.append("\n=== Recent Conversation (Sliding Window) ===")
220            for turn in recent:
221                if turn.content not in seen_contents:
222                    parts.append(f"[{turn.role}] {turn.content}")
223                    seen_contents.add(turn.content)
224
225        return "\n".join(parts) if parts else ""
226
227    def clear(self) -> None:
228        """Clear both sliding window and semantic memory stores."""
229        self.buffer.clear()
230        self.semantic.clear()
